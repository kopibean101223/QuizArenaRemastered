const WebSocket = require('ws');
const Redis = require('ioredis');

const wss = new WebSocket.Server({ port: 8080 });
const redisPub = new Redis();
const redisSub = new Redis();

// Map to track which clients belong to which battle: Map<wsClient, battleId>
const clientBattleMap = new Map();

redisSub.subscribe('battle_channel');

// Forward messages ONLY to clients in the relevant battle
redisSub.on('message', (channel, rawMessage) => {
  try {
    const data = JSON.parse(rawMessage);
    const { battleId } = data;

    wss.clients.forEach((client) => {
      if (
        client.readyState === WebSocket.OPEN &&
        clientBattleMap.get(client) === battleId
      ) {
        client.send(JSON.stringify(data));
      }
    });
  } catch (err) {
    console.error('Invalid JSON received:', err);
  }
});

wss.on('connection', (ws) => {
  ws.on('message', async (rawMessage) => {
    try {
      const data = JSON.parse(rawMessage.toString());

      // Action: Join a room
      if (data.type === 'JOIN_BATTLE') {
        clientBattleMap.set(ws, data.battleId);
        
        // Fetch room-specific history using a Redis List keyed by battleId
        const history = await redisPub.lrange(`battle:${data.battleId}:history`, 0, -1);
        ws.send(JSON.stringify({ type: 'HISTORY', history: history.map(JSON.parse) }));
        return;
      }

      // Action: Chat / Answer Submission
      if (data.type === 'BATTLE_ACTION') {
        // Store history under room-specific key
        await redisPub.rpush(`battle:${data.battleId}:history`, JSON.stringify(data));
        await redisPub.ltrim(`battle:${data.battleId}:history`, -100, -1);

        // Publish to single global Redis channel
        redisPub.publish('battle_channel', JSON.stringify(data));
      }
    } catch (err) {
      console.error('Failed to parse incoming payload:', err);
    }
  });

  ws.on('close', () => clientBattleMap.delete(ws));
});
