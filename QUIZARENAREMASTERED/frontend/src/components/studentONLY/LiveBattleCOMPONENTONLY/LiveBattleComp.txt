import React, { useState, useEffect, useRef } from "react";
import { useApp } from "../../context/AppContext";
import { StudentTopBar } from "../shared/StudentTopBar";
import {
  Crown, Zap, Star, Trophy, Flame, MessageCircle,
  CheckCircle2, Wifi, WifiOff, Volume2, VolumeX, Send,
} from "lucide-react";

import {
  C, OPTION_COLORS, AVATAR_COLORS, REACTIONS, QUESTION,
  INIT_PLAYERS, INIT_CHAT, Player, ChatMsg, Vote,
} from "./constants";
import { CountdownBar } from "./CountdownBar";
import { AnswerBtn } from "./AnswerBtn";
import { LeaderRow } from "./LeaderRow";
import { ChatBubble } from "./ChatBubble";

export function LiveBattle() {
  const { navigate } = useApp();
  const [timeLeft, setTimeLeft] = useState(QUESTION.timeLimit);
  const [selected, setSelected] = useState<number | null>(null);
  const [revealed, setRevealed] = useState(false);
  const [players, setPlayers] = useState<Player[]>(INIT_PLAYERS);
  const [chat, setChat] = useState<ChatMsg[]>(INIT_CHAT);
  const [chatInput, setChatInput] = useState("");
  const [votes, setVotes] = useState<Vote[]>([
    { option: 1, count: 2, voters: ["Ana R.", "Carlo B."] },
    { option: 3, count: 1, voters: ["Maria S."] },
  ]);
  const [myVote, setMyVote] = useState<number | null>(1);
  const [mode, setMode] = useState<"solo" | "discussion">("discussion");
  const [disconnected, setDisconnected] = useState(false);
  const [speedMode] = useState(true);
  const [reactionBursts, setReactionBursts] = useState<{ id: number; emoji: string; x: number; y: number }[]>([]);
  const [muted, setMuted] = useState(false);
  const [questionNum] = useState(QUESTION.number);
  const chatEndRef = useRef<HTMLDivElement>(null);

  const isLeader = players.find((p) => p.isMe)?.isLeader || false;
  const meAsLeader = players.find((p) => p.isLeader);

  // Timer
  useEffect(() => {
    if (revealed) return;
    if (timeLeft <= 0) {
      setRevealed(true);
      return;
    }
    const t = setTimeout(() => setTimeLeft((n) => n - 1), 1000);
    return () => clearTimeout(t);
  }, [timeLeft, revealed]);

  // Scroll chat
  useEffect(() => {
    chatEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [chat]);

  // Simulate enemy votes coming in
  useEffect(() => {
    const timers = [
      setTimeout(() => setVotes((v) => [...v, { option: 1, count: 1, voters: ["Ben A."] }]), 3000),
      setTimeout(
        () =>
          setVotes((v) =>
            v.map((x) =>
              x.option === 1
                ? { ...x, count: x.count + 1, voters: [...x.voters, "Lea F."] }
                : x
            )
          ),
        6000
      ),
    ];
    return () => timers.forEach(clearTimeout);
  }, []);

  function handleSelect(i: number) {
    if (revealed || selected !== null) return;
    setSelected(i);
    // Auto-reveal after 1.5s for solo
    if (mode === "solo") setTimeout(() => setRevealed(true), 1500);
  }

  function handleVote(i: number) {
    setMyVote(i);
    setVotes((v) => {
      const newV = v
        .filter((x) => x.voters.indexOf("You") < 0)
        .map((x) => ({
          ...x,
          voters: x.voters.filter((n) => n !== "You"),
          count: x.voters.includes("You") ? x.count - 1 : x.count,
        }));
      const existing = newV.find((x) => x.option === i);
      if (existing) {
        return newV.map((x) =>
          x.option === i ? { ...x, count: x.count + 1, voters: [...x.voters, "You"] } : x
        );
      }
      return [...newV, { option: i, count: 1, voters: ["You"] }];
    });
    setSelected(i);
  }

  function handleConfirmLeader() {
    setRevealed(true);
  }

  function sendChat() {
    if (!chatInput.trim()) return;
    setChat((c) => [
      ...c,
      {
        id: Date.now(),
        player: "You",
        initials: "ME",
        color: AVATAR_COLORS[0],
        text: chatInput.trim(),
        ts: "now",
      },
    ]);
    setChatInput("");
  }

  function fireReaction(emoji: string, e: React.MouseEvent) {
    const rect = (e.target as HTMLElement).getBoundingClientRect();
    const id = Date.now();
    setReactionBursts((r) => [...r, { id, emoji, x: rect.left, y: rect.top }]);
    setTimeout(() => setReactionBursts((r) => r.filter((x) => x.id !== id)), 1200);
  }

  // Vote percentages
  const totalVotes = votes.reduce((a, v) => a + v.count, 0);
  function voteFor(i: number) {
    const v = votes.find((x) => x.option === i);
    return totalVotes ? ((v?.count ?? 0) / totalVotes) * 100 : 0;
  }

  return (
    <>
      <StudentTopBar />
      <style>{`
        @keyframes timerPulse   { 0%,100%{transform:scale(1)} 50%{transform:scale(1.1)} }
        @keyframes rankPop      { 0%{transform:scale(0.9)} 60%{transform:scale(1.05)} 100%{transform:scale(1)} }
        @keyframes reactionFloat{ 0%{opacity:1;transform:translateY(0) scale(1)} 100%{opacity:0;transform:translateY(-80px) scale(1.5)} }
        @keyframes reconnecting { 0%,100%{opacity:1} 50%{opacity:0.5} }
        @keyframes floatA       { 0%,100%{transform:translateY(0)} 50%{transform:translateY(-8px)} }
        @keyframes correctBurst { 0%{transform:scale(0.8);opacity:0} 60%{transform:scale(1.06)} 100%{transform:scale(1);opacity:1} }
        @keyframes slideDown    { 0%{opacity:0;transform:translateY(-10px)} 100%{opacity:1;transform:translateY(0)} }
        @keyframes dotBlink     { 0%,100%{opacity:1} 50%{opacity:0} }
      `}</style>

      {/* Reaction bursts */}
      {reactionBursts.map((r) => (
        <div
          key={r.id}
          style={{
            position: "fixed",
            left: r.x,
            top: r.y,
            fontSize: 28,
            pointerEvents: "none",
            zIndex: 1000,
            animation: "reactionFloat 1.1s ease-out forwards",
          }}
        >
          {r.emoji}
        </div>
      ))}

      {/* Reconnecting banner */}
      {disconnected && (
        <div
          style={{
            position: "fixed",
            top: 48,
            left: 0,
            right: 0,
            zIndex: 900,
            background: "rgba(255,71,87,0.95)",
            padding: "10px 20px",
            display: "flex",
            alignItems: "center",
            justifyContent: "center",
            gap: 10,
            animation: "slideDown 0.3s ease-out",
          }}
        >
          <WifiOff
            size={16}
            color="#fff"
            strokeWidth={2.5}
            style={{ animation: "reconnecting 1s ease-in-out infinite" }}
          />
          <span style={{ fontFamily: "Manrope, sans-serif", fontSize: 13, fontWeight: 700, color: "#fff" }}>
            Reconnecting to battle…
          </span>
          <div style={{ display: "flex", gap: 3 }}>
            {[0, 1, 2].map((i) => (
              <span
                key={i}
                style={{
                  width: 5,
                  height: 5,
                  borderRadius: "50%",
                  background: "rgba(255,255,255,0.6)",
                  display: "inline-block",
                  animation: `dotBlink 1s ease-in-out ${i * 0.2}s infinite`,
                }}
              />
            ))}
          </div>
          <button
            type="button"
            onClick={() => setDisconnected(false)}
            style={{
              marginLeft: 20,
              background: "rgba(255,255,255,0.2)",
              border: "none",
              borderRadius: 8,
              padding: "4px 12px",
              color: "#fff",
              fontSize: 12,
              fontWeight: 700,
              fontFamily: "Manrope, sans-serif",
              cursor: "pointer",
            }}
          >
            Dismiss
          </button>
        </div>
      )}

      <div
        style={{
          minHeight: "100vh",
          background: `radial-gradient(ellipse at 20% 20%, rgba(91,61,246,0.15) 0%, transparent 50%),
        radial-gradient(ellipse at 80% 80%, rgba(255,107,74,0.1) 0%, transparent 50%), ${C.navy}`,
          display: "flex",
          flexDirection: "column",
          overflow: "hidden",
          paddingTop: 48,
        }}
      >
        {/* ── TOP BAR ── */}
        <div
          style={{
            padding: "12px 20px",
            display: "flex",
            alignItems: "center",
            gap: 14,
            background: "rgba(0,0,0,0.25)",
            borderBottom: "1.5px solid rgba(255,255,255,0.06)",
            flexShrink: 0,
          }}
        >
          <div style={{ display: "flex", alignItems: "center", gap: 10, flexShrink: 0 }}>
            <div
              style={{
                width: 36,
                height: 36,
                borderRadius: 11,
                background: C.indigo,
                display: "flex",
                alignItems: "center",
                justifyContent: "center",
                boxShadow: "0 3px 12px rgba(91,61,246,0.4)",
              }}
            >
              <Trophy fill={C.yellow} color="transparent" size={18} />
            </div>
            <div>
              <p
                style={{
                  fontFamily: "Fredoka, sans-serif",
                  fontSize: 13,
                  fontWeight: 600,
                  color: "rgba(255,255,255,0.5)",
                  margin: 0,
                  lineHeight: 1,
                }}
              >
                Question
              </p>
              <p
                style={{
                  fontFamily: "Fredoka, sans-serif",
                  fontSize: 19,
                  fontWeight: 700,
                  color: "#fff",
                  margin: 0,
                  lineHeight: 1,
                }}
              >
                {questionNum} <span style={{ color: "rgba(255,255,255,0.3)" }}>/ {QUESTION.total}</span>
              </p>
            </div>
          </div>

          <CountdownBar timeLeft={timeLeft} timeLimit={QUESTION.timeLimit} />

          <div style={{ display: "flex", alignItems: "center", gap: 8, flexShrink: 0 }}>
            {speedMode && (
              <div
                style={{
                  display: "flex",
                  alignItems: "center",
                  gap: 5,
                  background: "rgba(255,201,60,0.15)",
                  border: "1.5px solid rgba(255,201,60,0.35)",
                  borderRadius: 20,
                  padding: "5px 12px",
                }}
              >
                <Zap size={13} fill={C.yellow} color="transparent" />
                <span
                  style={{
                    fontFamily: "Manrope, sans-serif",
                    fontSize: 12,
                    fontWeight: 800,
                    color: C.yellow,
                    letterSpacing: "0.05em",
                  }}
                >
                  SPEED MODE
                </span>
              </div>
            )}
            <div
              style={{
                display: "flex",
                alignItems: "center",
                gap: 5,
                background: "rgba(255,255,255,0.06)",
                border: "1.5px solid rgba(255,255,255,0.1)",
                borderRadius: 20,
                padding: "5px 12px",
              }}
            >
              <Star size={12} fill={C.yellow} color="transparent" />
              <span style={{ fontFamily: "Fredoka, sans-serif", fontSize: 14, fontWeight: 700, color: "#fff" }}>
                {QUESTION.points} pts
              </span>
            </div>

            <div
              style={{
                display: "flex",
                background: "rgba(255,255,255,0.07)",
                borderRadius: 20,
                padding: "3px 4px",
                gap: 3,
              }}
            >
              {(
                [
                  ["solo", "⚡ Solo"],
                  ["discussion", "👥 Team"],
                ] as const
              ).map(([v, l]) => (
                <button
                  key={v}
                  type="button"
                  onClick={() => setMode(v)}
                  style={{
                    background: mode === v ? C.indigo : "transparent",
                    border: "none",
                    borderRadius: 16,
                    padding: "4px 10px",
                    fontFamily: "Manrope, sans-serif",
                    fontSize: 11,
                    fontWeight: 700,
                    color: mode === v ? "#fff" : "rgba(255,255,255,0.4)",
                    cursor: "pointer",
                    transition: "all 0.15s",
                  }}
                >
                  {l}
                </button>
              ))}
            </div>

            <button
              type="button"
              onClick={() => setMuted((v) => !v)}
              style={{
                background: "rgba(255,255,255,0.06)",
                border: "1.5px solid rgba(255,255,255,0.1)",
                borderRadius: "50%",
                width: 34,
                height: 34,
                display: "flex",
                alignItems: "center",
                justifyContent: "center",
                cursor: "pointer",
                color: "rgba(255,255,255,0.45)",
              }}
            >
              {muted ? <VolumeX size={15} strokeWidth={2} /> : <Volume2 size={15} strokeWidth={2} />}
            </button>
            <button
              type="button"
              onClick={() => setDisconnected((v) => !v)}
              style={{
                background: "rgba(255,255,255,0.06)",
                border: "1.5px solid rgba(255,255,255,0.1)",
                borderRadius: "50%",
                width: 34,
                height: 34,
                display: "flex",
                alignItems: "center",
                justifyContent: "center",
                cursor: "pointer",
                color: "rgba(255,255,255,0.45)",
              }}
            >
              <Wifi size={15} strokeWidth={2} />
            </button>
          </div>
        </div>

        {/* ── MAIN BODY ── */}
        <div style={{ flex: 1, display: "flex", gap: 0, overflow: "hidden", minHeight: 0 }}>
          {/* ── CENTER: QUESTION + ANSWERS ── */}
          <div
            style={{
              flex: 1,
              display: "flex",
              flexDirection: "column",
              padding: "20px 20px 16px",
              overflow: "hidden",
              minWidth: 0,
            }}
          >
            {/* Question card */}
            <div
              style={{
                background: "rgba(255,255,255,0.05)",
                border: "1.5px solid rgba(255,255,255,0.08)",
                borderRadius: 24,
                padding: "22px 26px",
                marginBottom: 16,
                flexShrink: 0,
                boxShadow: "0 8px 32px rgba(0,0,0,0.3), inset 0 1px 0 rgba(255,255,255,0.06)",
              }}
            >
              <div style={{ display: "flex", alignItems: "center", gap: 8, marginBottom: 12, flexWrap: "wrap" }}>
                <span
                  style={{
                    background: C.indigoLight,
                    border: "1.5px solid rgba(91,61,246,0.3)",
                    borderRadius: 8,
                    padding: "3px 10px",
                    fontFamily: "Manrope, sans-serif",
                    fontSize: 11,
                    fontWeight: 700,
                    color: "#A08FFF",
                  }}
                >
                  {QUESTION.subject}
                </span>
                <span
                  style={{
                    fontFamily: "Manrope, sans-serif",
                    fontSize: 11,
                    fontWeight: 700,
                    color: "rgba(255,255,255,0.35)",
                    textTransform: "uppercase",
                    letterSpacing: "0.07em",
                  }}
                >
                  Multiple Choice
                </span>
              </div>
              <p
                style={{
                  fontFamily: "Fredoka, sans-serif",
                  fontSize: 26,
                  fontWeight: 700,
                  color: "#fff",
                  margin: 0,
                  lineHeight: 1.35,
                }}
              >
                {QUESTION.text}
              </p>
            </div>

            {/* Answer options */}
            <div style={{ display: "flex", flexDirection: "column", gap: 10, flex: 1, overflow: "hidden" }}>
              {QUESTION.options.map((opt, i) => (
                <AnswerBtn
                  key={i}
                  index={i}
                  text={opt}
                  selected={mode === "discussion" ? myVote === i : selected === i}
                  revealed={revealed}
                  isCorrect={i === QUESTION.correct}
                  disabled={revealed || (mode === "solo" && selected !== null)}
                  onClick={() => (mode === "discussion" ? handleVote(i) : handleSelect(i))}
                  votePct={mode === "discussion" ? voteFor(i) : undefined}
                />
              ))}
            </div>

            {/* Reveal result feedback */}
            {revealed && (
              <div
                style={{
                  marginTop: 14,
                  padding: "14px 18px",
                  borderRadius: 18,
                  background:
                    selected === QUESTION.correct ? "rgba(46,212,122,0.15)" : "rgba(255,71,87,0.12)",
                  border: `2px solid ${selected === QUESTION.correct ? C.green : C.red}`,
                  display: "flex",
                  alignItems: "center",
                  gap: 12,
                  flexShrink: 0,
                  animation: "correctBurst 0.5s cubic-bezier(0.34,1.56,0.64,1)",
                }}
              >
                <span style={{ fontSize: 28, flexShrink: 0 }}>
                  {selected === QUESTION.correct ? "🎉" : "❌"}
                </span>
                <div>
                  <p
                    style={{
                      fontFamily: "Fredoka, sans-serif",
                      fontSize: 20,
                      fontWeight: 700,
                      color: selected === QUESTION.correct ? C.green : C.red,
                      margin: 0,
                    }}
                  >
                    {selected === QUESTION.correct ? "Correct! +" + QUESTION.points + " pts" : "Wrong Answer"}
                  </p>
                  <p style={{ fontFamily: "Manrope, sans-serif", fontSize: 13, color: "rgba(255,255,255,0.55)", margin: 0 }}>
                    {selected === null ? "Time's up! " : ""}
                    The answer is: <strong style={{ color: "#fff" }}>{QUESTION.options[QUESTION.correct]}</strong>
                  </p>
                </div>
                {revealed && (
                  <button
                    type="button"
                    onClick={() => navigate("results")}
                    style={{
                      marginLeft: "auto",
                      background: C.indigo,
                      border: "none",
                      borderRadius: 12,
                      padding: "8px 16px",
                      fontFamily: "Manrope, sans-serif",
                      fontSize: 12,
                      fontWeight: 700,
                      color: "#fff",
                      cursor: "pointer",
                      flexShrink: 0,
                    }}
                  >
                    View Results →
                  </button>
                )}
              </div>
            )}
          </div>

          {/* ── RIGHT: LEADERBOARD ── */}
          <div
            style={{
              width: 240,
              minWidth: 240,
              background: "rgba(0,0,0,0.2)",
              borderLeft: "1.5px solid rgba(255,255,255,0.06)",
              display: "flex",
              flexDirection: "column",
              overflow: "hidden",
            }}
          >
            <div
              style={{
                padding: "14px 14px 10px",
                borderBottom: "1.5px solid rgba(255,255,255,0.06)",
                flexShrink: 0,
              }}
            >
              <div style={{ display: "flex", alignItems: "center", gap: 7 }}>
                <Trophy fill={C.yellow} color="transparent" size={16} />
                <span style={{ fontFamily: "Fredoka, sans-serif", fontSize: 17, fontWeight: 700, color: "#fff" }}>
                  Leaderboard
                </span>
              </div>
              <p
                style={{
                  fontFamily: "Manrope, sans-serif",
                  fontSize: 11,
                  color: "rgba(255,255,255,0.3)",
                  margin: "3px 0 0",
                  fontWeight: 500,
                }}
              >
                {players.length} players · live
              </p>
            </div>
            <div style={{ flex: 1, overflowY: "auto", padding: "8px 10px", display: "flex", flexDirection: "column", gap: 5 }}>
              {[...players]
                .sort((a, b) => a.rank - b.rank)
                .map((p) => (
                  <LeaderRow key={p.id} player={p} />
                ))}
            </div>
          </div>
        </div>

        {/* ── DISCUSSION MODE PANEL ── */}
        {mode === "discussion" && (
          <div
            style={{
              height: 220,
              background: "rgba(0,0,0,0.3)",
              borderTop: "1.5px solid rgba(255,255,255,0.07)",
              display: "flex",
              gap: 0,
              flexShrink: 0,
              animation: "slideDown 0.3s ease-out",
            }}
          >
            {/* Chat feed */}
            <div
              style={{
                flex: 1,
                display: "flex",
                flexDirection: "column",
                borderRight: "1.5px solid rgba(255,255,255,0.06)",
                overflow: "hidden",
              }}
            >
              <div
                style={{
                  padding: "8px 14px 6px",
                  borderBottom: "1px solid rgba(255,255,255,0.05)",
                  display: "flex",
                  alignItems: "center",
                  justifyContent: "space-between",
                  flexShrink: 0,
                }}
              >
                <div style={{ display: "flex", alignItems: "center", gap: 6 }}>
                  <MessageCircle size={13} color="rgba(255,255,255,0.4)" strokeWidth={2} />
                  <span
                    style={{
                      fontFamily: "Manrope, sans-serif",
                      fontSize: 12,
                      fontWeight: 700,
                      color: "rgba(255,255,255,0.4)",
                    }}
                  >
                    Team Discussion
                  </span>
                </div>
                {meAsLeader && (
                  <div
                    style={{
                      display: "flex",
                      alignItems: "center",
                      gap: 5,
                      background: "rgba(255,201,60,0.15)",
                      border: "1.5px solid rgba(255,201,60,0.3)",
                      borderRadius: 20,
                      padding: "3px 10px",
                      animation: "floatA 2.5s ease-in-out infinite",
                    }}
                  >
                    <Crown size={11} fill={C.yellow} color="transparent" />
                    <span style={{ fontFamily: "Manrope, sans-serif", fontSize: 10, fontWeight: 800, color: C.yellow }}>
                      {meAsLeader.name} leads
                    </span>
                  </div>
                )}
              </div>

              <div style={{ flex: 1, overflowY: "auto", padding: "8px 12px", display: "flex", flexDirection: "column", gap: 8 }}>
                {chat.map((m) => (
                  <ChatBubble key={m.id} msg={m} />
                ))}
                <div ref={chatEndRef} />
              </div>

              <div
                style={{
                  padding: "8px 10px",
                  borderTop: "1px solid rgba(255,255,255,0.05)",
                  display: "flex",
                  gap: 7,
                  alignItems: "center",
                  flexShrink: 0,
                }}
              >
                <div style={{ display: "flex", gap: 4 }}>
                  {REACTIONS.map((emoji) => (
                    <button
                      key={emoji}
                      type="button"
                      onClick={(e) => {
                        fireReaction(emoji, e);
                        setChat((c) => [
                          ...c,
                          {
                            id: Date.now(),
                            player: "You",
                            initials: "ME",
                            color: AVATAR_COLORS[0],
                            text: emoji,
                            ts: "now",
                          },
                        ]);
                      }}
                      style={{
                        width: 28,
                        height: 28,
                        borderRadius: 8,
                        background: "rgba(255,255,255,0.06)",
                        border: "1px solid rgba(255,255,255,0.08)",
                        fontSize: 14,
                        cursor: "pointer",
                        display: "flex",
                        alignItems: "center",
                        justifyContent: "center",
                        transition: "transform 0.15s",
                      }}
                    >
                      {emoji}
                    </button>
                  ))}
                </div>
                <input
                  value={chatInput}
                  onChange={(e) => setChatInput(e.target.value)}
                  onKeyDown={(e) => {
                    if (e.key === "Enter") sendChat();
                  }}
                  placeholder="Type to discuss…"
                  style={{
                    flex: 1,
                    background: "rgba(255,255,255,0.06)",
                    border: "1.5px solid rgba(255,255,255,0.1)",
                    borderRadius: 10,
                    padding: "7px 12px",
                    fontFamily: "Manrope, sans-serif",
                    fontSize: 13,
                    fontWeight: 500,
                    color: "#fff",
                    outline: "none",
                  }}
                />
                <button
                  type="button"
                  onClick={sendChat}
                  style={{
                    width: 32,
                    height: 32,
                    background: C.indigo,
                    border: "none",
                    borderRadius: 9,
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "center",
                    cursor: "pointer",
                    flexShrink: 0,
                    boxShadow: "0 2px 8px rgba(91,61,246,0.4)",
                  }}
                >
                  <Send size={14} color="#fff" strokeWidth={2.5} />
                </button>
              </div>
            </div>

            {/* Vote panel */}
            <div
              style={{
                width: 260,
                display: "flex",
                flexDirection: "column",
                borderRight: "1.5px solid rgba(255,255,255,0.06)",
                overflow: "hidden",
              }}
            >
              <div style={{ padding: "8px 14px 6px", borderBottom: "1px solid rgba(255,255,255,0.05)", flexShrink: 0 }}>
                <span style={{ fontFamily: "Manrope, sans-serif", fontSize: 12, fontWeight: 700, color: "rgba(255,255,255,0.4)" }}>
                  Team Votes
                </span>
              </div>
              <div style={{ flex: 1, overflowY: "auto", padding: "8px 12px", display: "flex", flexDirection: "column", gap: 6 }}>
                {QUESTION.options.map((opt, i) => {
                  const vpct = voteFor(i);
                  const vdata = votes.find((v) => v.option === i);
                  const isMyVote = myVote === i;
                  return (
                    <button
                      key={i}
                      type="button"
                      onClick={() => !revealed && handleVote(i)}
                      style={{
                        background: isMyVote ? "rgba(91,61,246,0.2)" : "rgba(255,255,255,0.04)",
                        border: `1.5px solid ${isMyVote ? "rgba(91,61,246,0.5)" : "rgba(255,255,255,0.07)"}`,
                        borderRadius: 12,
                        padding: "7px 10px",
                        cursor: revealed ? "default" : "pointer",
                        display: "flex",
                        alignItems: "center",
                        gap: 8,
                        transition: "all 0.15s",
                        width: "100%",
                      }}
                    >
                      <span
                        style={{
                          width: 22,
                          height: 22,
                          borderRadius: 7,
                          flexShrink: 0,
                          background: OPTION_COLORS[i].base + "33",
                          display: "flex",
                          alignItems: "center",
                          justifyContent: "center",
                          fontFamily: "Fredoka, sans-serif",
                          fontSize: 13,
                          fontWeight: 700,
                          color: OPTION_COLORS[i].base,
                        }}
                      >
                        {["A", "B", "C", "D"][i]}
                      </span>
                      <span
                        style={{
                          fontFamily: "Manrope, sans-serif",
                          fontSize: 12,
                          fontWeight: 600,
                          color: "rgba(255,255,255,0.6)",
                          flex: 1,
                          textAlign: "left",
                          overflow: "hidden",
                          textOverflow: "ellipsis",
                          whiteSpace: "nowrap",
                        }}
                      >
                        {opt}
                      </span>
                      <div style={{ display: "flex", alignItems: "center", gap: 5, flexShrink: 0 }}>
                        {vdata && (
                          <div style={{ display: "flex" }}>
                            {vdata.voters.slice(0, 3).map((voter, vi) => (
                              <div
                                key={vi}
                                style={{
                                  width: 16,
                                  height: 16,
                                  borderRadius: "50%",
                                  background: AVATAR_COLORS[vi],
                                  border: "1.5px solid rgba(0,0,0,0.4)",
                                  marginLeft: vi > 0 ? -5 : 0,
                                  fontSize: 7,
                                  fontWeight: 800,
                                  display: "flex",
                                  alignItems: "center",
                                  justifyContent: "center",
                                  color: "#fff",
                                  fontFamily: "Manrope, sans-serif",
                                  zIndex: 3 - vi,
                                  position: "relative",
                                }}
                              >
                                {voter[0]}
                              </div>
                            ))}
                          </div>
                        )}
                        <span
                          style={{
                            fontFamily: "Fredoka, sans-serif",
                            fontSize: 13,
                            fontWeight: 700,
                            color: isMyVote ? "#A08FFF" : "rgba(255,255,255,0.35)",
                          }}
                        >
                          {Math.round(vpct)}%
                        </span>
                      </div>
                    </button>
                  );
                })}
              </div>
            </div>

            {/* Leader confirm panel */}
            <div
              style={{
                width: 200,
                display: "flex",
                flexDirection: "column",
                justifyContent: "center",
                alignItems: "center",
                padding: "16px 14px",
                gap: 12,
              }}
            >
              <div style={{ textAlign: "center" }}>
                <Crown size={22} fill={C.yellow} color="transparent" style={{ animation: "floatA 2s ease-in-out infinite" }} />
                <p style={{ fontFamily: "Fredoka, sans-serif", fontSize: 16, fontWeight: 700, color: "#fff", margin: "6px 0 2px" }}>
                  Leader Decision
                </p>
                <p style={{ fontFamily: "Manrope, sans-serif", fontSize: 11, color: "rgba(255,255,255,0.4)", margin: 0, lineHeight: 1.5 }}>
                  {meAsLeader ? `${meAsLeader.name} leads the team` : "Waiting for leader to confirm…"}
                </p>
              </div>

              {meAsLeader && (
                <div style={{ display: "flex", alignItems: "center", gap: 7 }}>
                  <div style={{ position: "relative" }}>
                    <div
                      style={{
                        width: 36,
                        height: 36,
                        borderRadius: "50%",
                        background: meAsLeader.color,
                        display: "flex",
                        alignItems: "center",
                        justifyContent: "center",
                        fontFamily: "Manrope, sans-serif",
                        fontSize: 12,
                        fontWeight: 800,
                        color: "#fff",
                        border: "2px solid rgba(255,201,60,0.5)",
                      }}
                    >
                      {meAsLeader.initials}
                    </div>
                    <Crown
                      size={14}
                      fill={C.yellow}
                      color="transparent"
                      style={{ position: "absolute", top: -8, left: "50%", transform: "translateX(-50%)" }}
                    />
                  </div>
                  <span style={{ fontFamily: "Manrope, sans-serif", fontSize: 12, fontWeight: 700, color: "rgba(255,255,255,0.7)" }}>
                    {meAsLeader.name}
                  </span>
                </div>
              )}

              {meAsLeader && !revealed && (
                <button
                  type="button"
                  onClick={handleConfirmLeader}
                  disabled={myVote === null}
                  style={{
                    width: "100%",
                    background:
                      myVote !== null
                        ? `linear-gradient(135deg,${C.green},${C.greenDeep})`
                        : "rgba(46,212,122,0.2)",
                    border: "none",
                    borderRadius: 14,
                    padding: "11px 0",
                    fontFamily: "Fredoka, sans-serif",
                    fontSize: 17,
                    fontWeight: 700,
                    color: myVote !== null ? "#fff" : "rgba(255,255,255,0.3)",
                    cursor: myVote !== null ? "pointer" : "default",
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "center",
                    gap: 7,
                    boxShadow: myVote !== null ? "0 4px 16px rgba(46,212,122,0.35)" : "none",
                    transition: "all 0.2s",
                  }}
                >
                  <CheckCircle2 size={17} strokeWidth={2.5} />
                  Confirm Answer
                </button>
              )}
              {revealed && (
                <div
                  style={{
                    display: "flex",
                    alignItems: "center",
                    gap: 6,
                    background: "rgba(46,212,122,0.12)",
                    borderRadius: 12,
                    padding: "8px 12px",
                    border: "1.5px solid rgba(46,212,122,0.25)",
                  }}
                >
                  <CheckCircle2 size={14} color={C.green} strokeWidth={2.5} />
                  <span style={{ fontFamily: "Manrope, sans-serif", fontSize: 12, fontWeight: 700, color: C.green }}>
                    Answer locked!
                  </span>
                </div>
              )}
              {!meAsLeader && !revealed && (
                <div
                  style={{
                    padding: "8px 10px",
                    background: "rgba(255,255,255,0.04)",
                    border: "1.5px solid rgba(255,255,255,0.07)",
                    borderRadius: 12,
                    textAlign: "center",
                  }}
                >
                  <div style={{ display: "flex", justifyContent: "center", gap: 3, marginBottom: 5 }}>
                    {[0, 1, 2].map((i) => (
                      <span
                        key={i}
                        style={{
                          width: 5,
                          height: 5,
                          borderRadius: "50%",
                          background: C.yellow,
                          display: "inline-block",
                          animation: `dotBlink 1s ease-in-out ${i * 0.25}s infinite`,
                        }}
                      />
                    ))}
                  </div>
                  <span style={{ fontFamily: "Manrope, sans-serif", fontSize: 11, color: "rgba(255,255,255,0.35)", fontWeight: 600 }}>
                    Waiting for leader…
                  </span>
                </div>
              )}
            </div>
          </div>
        )}
      </div>
    </>
  );
}
